## 1. 测试概述

### 1.1 测试目标

| 目标 | 说明 |
|------|------|
| **正确性** | 数学计算、知识图谱、数据模型等核心逻辑正确无误 |
| **稳定性** | API 端点、会话流程、状态机流转稳定可靠 |
| **安全性** | 内容过滤、危机检测、数据安全等安全机制有效 |
| **教学质量** | LLM 解题正确、人设一致、红线遵守 |
| **回归保护** | 代码修改后不引入新 bug |

### 1.2 测试范围

| 层级 | 覆盖内容 | 工具 |
|------|---------|------|
| 单元测试 | 各模块独立功能验证 | pytest |
| 集成测试 | API 端点、会话流程、状态机 | pytest + FastAPI TestClient |
| LLM 评估 | 解题正确性、人设一致性、内容安全 | 自定义测试运行器 |
| 手动测试 | UI 交互、用户体验 | 人工验证 |

### 1.3 测试策略

- **核心模块必须有测试**：知识图谱、数据模型、存储层、内容过滤、危机检测、计算验证
- **测试优先覆盖边界条件**：空输入、异常值、边界情况
- **LLM 相关测试标记为慢速**：需要 API Key，默认不运行，用 `--run-slow` 触发
- **测试数据使用临时目录**：不污染真实学生数据
- **每个测试独立运行**：不依赖其他测试的状态

### 3.1 知识图谱测试（test_syllabus.py）

**测试目的**：验证知识图谱的完整性、一致性和无环性。

**测试用例**：

| 编号 | 测试名称 | 预期结果 |
|------|---------|---------|
| T-SYL-01 | 无重复 ID | 所有节点 ID 唯一 |
| T-SYL-02 | 前置依赖存在 | 所有 prerequisites 指向存在的节点 |
| T-SYL-03 | 无循环依赖 | 依赖图中无环 |
| T-SYL-04 | 所有节点有章节 | 每个节点 chapter 非空 |
| T-SYL-05 | 难度范围合法 | difficulty 在 1-5 之间 |
| T-SYL-06 | get_node 正确 | 按 ID 能正确获取节点 |
| T-SYL-07 | all_nodes 返回全部 | 节点数量与定义一致 |
| T-SYL-08 | 认知门槛字段存在 | required_cognitive_stage 有值 |
| T-SYL-09 | 思维模型字段存在 | common_thinking_models 有值或空列表 |
| T-SYL-10 | validate 方法可用 | 调用 validate 返回错误列表 |

**代码示例**：

```python
"""知识图谱单元测试。"""
import pytest
from backend.knowledge import syllabus


class TestSyllabusIntegrity:
    def test_no_duplicate_ids(self):
        """T-SYL-01: 所有节点 ID 唯一。"""
        ids = [n.id for n in syllabus.all_nodes()]
        assert len(ids) == len(set(ids)), f"存在重复ID: {[x for x in ids if ids.count(x) > 1]}"

    def test_prerequisites_exist(self):
        """T-SYL-02: 所有前置依赖指向存在的节点。"""
        all_ids = {n.id for n in syllabus.all_nodes()}
        for n in syllabus.all_nodes():
            for pre in n.prerequisites:
                assert pre in all_ids, f"节点 {n.id} 的前置 {pre} 不存在"

    def test_no_circular_dependencies(self):
        """T-SYL-03: 依赖图中无环（DFS 检测）。"""
        all_ids = {n.id for n in syllabus.all_nodes()}
        adj = {n.id: n.prerequisites for n in syllabus.all_nodes()}

        WHITE, GRAY, BLACK = 0, 1, 2
        color = {nid: WHITE for nid in all_ids}

        def dfs(node):
            color[node] = GRAY
            for neighbor in adj.get(node, []):
                if color[neighbor] == GRAY:
                    return True  # 发现环
                if color[neighbor] == WHITE and dfs(neighbor):
                    return True
            color[node] = BLACK
            return False

        for nid in all_ids:
            if color[nid] == WHITE:
                assert not dfs(nid), f"节点 {nid} 所在子图存在循环依赖"

    def test_all_nodes_have_chapter(self):
        """T-SYL-04: 每个节点 chapter 非空。"""
        for n in syllabus.all_nodes():
            assert n.chapter, f"节点 {n.id} 没有章节"

    def test_difficulty_range(self):
        """T-SYL-05: difficulty 在 1-5 之间。"""
        for n in syllabus.all_nodes():
            assert 1 <= n.difficulty <= 5, f"节点 {n.id} 难度 {n.difficulty} 超出范围"

    def test_get_node_correct(self):
        """T-SYL-06: 按 ID 能正确获取节点。"""
        for n in syllabus.all_nodes():
            found = syllabus.get_node(n.id)
            assert found is not None, f"get_node({n.id}) 返回 None"
            assert found.id == n.id

    def test_get_node_nonexistent(self):
        """get_node 对不存在的 ID 返回 None。"""
        assert syllabus.get_node("nonexistent_id_xyz") is None

    def test_all_nodes_count(self):
        """T-SYL-07: all_nodes 返回非空列表。"""
        nodes = syllabus.all_nodes()
        assert len(nodes) > 0
        assert isinstance(nodes, list)
```

**代码示例**：

```python
"""存储层单元测试。"""
import pytest
from backend.services.storage import StudentStorage


class TestStudentStorage:
    def test_create_student(self, tmp_student_dir):
        """T-STO-01: 创建学生，文件已保存。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        s = storage.create("测试学生", 6)
        assert s.id is not None
        assert s.name == "测试学生"
        assert s.grade == 6
        # 验证文件存在
        assert (tmp_student_dir / f"{s.id}.json").exists()

    def test_load_existing_student(self, tmp_student_dir, saved_student):
        """T-STO-02: 加载存在的学生。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        loaded = storage.load(saved_student.id)
        assert loaded is not None
        assert loaded.id == saved_student.id
        assert loaded.name == saved_student.name

    def test_load_nonexistent_student(self, tmp_student_dir):
        """T-STO-03: 加载不存在的学生返回 None。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        assert storage.load("nonexistent_id") is None

    def test_save_persistence(self, tmp_student_dir, saved_student):
        """T-STO-04: 修改后保存，重新加载数据一致。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        saved_student.mastery["test_topic"] = 0.75
        saved_student.badges.append("🔗 不断链7天")
        storage.save(saved_student)

        loaded = storage.load(saved_student.id)
        assert loaded.mastery["test_topic"] == 0.75
        assert "🔗 不断链7天" in loaded.badges

    def test_delete_student(self, tmp_student_dir, saved_student):
        """T-STO-05: 删除学生后文件不存在。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        path = tmp_student_dir / f"{saved_student.id}.json"
        assert path.exists()
        path.unlink()
        assert storage.load(saved_student.id) is None

    def test_list_all_students(self, tmp_student_dir):
        """T-STO-06: list_all 返回所有已创建学生。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        s1 = storage.create("学生A", 6)
        s2 = storage.create("学生B", 7)
        all_students = storage.list_all()
        assert len(all_students) == 2
        names = {s["name"] for s in all_students}
        assert "学生A" in names
        assert "学生B" in names

    def test_unique_ids(self, tmp_student_dir):
        """T-STO-08: 两次 create 生成不同 ID。"""
        storage = StudentStorage()
        storage.STUDENTS_DIR = tmp_student_dir
        s1 = storage.create("学生A", 6)
        s2 = storage.create("学生B", 6)
        assert s1.id != s2.id
```

### 3.4 内容过滤测试（test_content_filter.py）

**测试目的**：验证 LLM 输出内容过滤的正确性。

**测试用例**：

| 编号 | 测试名称 | 输入 | 预期结果 |
|------|---------|------|---------|
| T-CF-01 | 正常内容通过 | "我们来解这道方程" | 不触发，原文返回 |
| T-CF-02 | 暴力内容拦截 | "这道题用暴力解法" | 触发（注意：可能误判，需验证） |
| T-CF-03 | 敏感话题-政治 | "我们来讨论政治" | 触发，返回政治引导语 |
| T-CF-04 | 敏感话题-宗教 | "佛教和基督教哪个好" | 触发，返回宗教引导语 |
| T-CF-05 | 空输入 | "" | 不触发，返回空 |
| T-CF-06 | 数学正常内容 | "2x + 5 = 13，x = 4" | 不触发 |
| T-CF-07 | filter_llm_output 便捷函数 | 正常文本 | 返回 (text, False) |

**代码示例**：

```python
"""内容过滤单元测试。"""
import pytest
from backend.services.content_filter import ContentFilter, filter_llm_output


class TestContentFilter:
    def setup_method(self):
        self.cf = ContentFilter()

    def test_normal_content_passes(self):
        """T-CF-01: 正常内容不触发过滤。"""
        text = "我们来解这道方程，先把常数项移到右边。"
        result, triggered = self.cf.filter_output(text)
        assert triggered is False
        assert result == text

    def test_politics_sensitive(self):
        """T-CF-03: 政治敏感话题触发引导。"""
        text = "我们来讨论一下政治问题"
        result, triggered = self.cf.filter_output(text)
        assert triggered is True
        assert "政治" in result or "数学" in result

    def test_religion_sensitive(self):
        """T-CF-04: 宗教敏感话题触发引导。"""
        text = "佛教和基督教哪个更好"
        result, triggered = self.cf.filter_output(text)
        assert triggered is True

    def test_empty_input(self):
        """T-CF-05: 空输入不触发。"""
        result, triggered = self.cf.filter_output("")
        assert triggered is False
        assert result == ""
```

| T-SM-06 | WRAP_UP→DONE | WRAP_UP | 结束会话 | DONE/重置 |

**代码示例**：

```python
"""状态机集成测试。"""
import pytest
from backend.models.student import SessionState
from backend.agent.assessment import apply_eval


VALID_STATES = [
    "GREETING", "MOOD_CHECKIN", "MODE_SELECT", "PLAN",
    "BLIND_SPOT", "CORE_DERIVE", "EXAMPLE_CHECK", "ERROR_REVIEW",
    "OPTIONAL_VARIANT", "QUICK_REVIEW", "FIX_ONE_ERROR",
    "WEEKEND_CLEAR", "WRAP_UP", "DONE",
]


class TestStateMachine:
    def test_initial_state(self):
        """初始状态为 GREETING。"""
        s = SessionState()
        assert s.state == "GREETING"

    def test_mode_a_transition(self):
        """T-SM-02: 选A模式进入 BLIND_SPOT。"""
        from backend.agent.chat import select_mode
        from backend.models.student import Student
        s = Student(id="test", name="测试")
        s.current_session.state = "MODE_SELECT"
        select_mode(s, "A")
        assert s.current_session.state == "BLIND_SPOT"

    def test_mode_b_transition(self):
        """T-SM-03: 选B模式进入 QUICK_REVIEW。"""
        from backend.agent.chat import select_mode
        from backend.models.student import Student
        s = Student(id="test", name="测试")
        s.current_session.state = "MODE_SELECT"
        select_mode(s, "B")
        assert s.current_session.state == "QUICK_REVIEW"

    def test_invalid_state_ignored(self, sample_student):
        """T-SM-04: 非法状态值不改变当前状态。"""
        sample_student.current_session.state = "GREETING"
        apply_eval(sample_student, {"state": "INVALID_STATE", "mastery_updates": {}})
        assert sample_student.current_session.state == "GREETING"

    def test_valid_state_transition(self, sample_student):
        """T-SM-05: 评估块合法更新状态。"""
        sample_student.current_session.state = "BLIND_SPOT"
        apply_eval(sample_student, {"state": "CORE_DERIVE", "mastery_updates": {}})
        assert sample_student.current_session.state == "CORE_DERIVE"

    def test_all_valid_states_defined(self):
        """所有状态在合法列表中。"""
        # 这个测试验证状态常量的完整性
        for state in VALID_STATES:
            assert isinstance(state, str)
            assert len(state) > 0
```

```python
"""人设一致性测试（需要 API Key，标记为 slow）。"""
import pytest
from backend.services.llm import simple_chat
from backend.agent.persona import PERSONA_CORE


pytestmark = pytest.mark.slow


PERSONA_TESTS = [
    {
        "id": "T-PC-01",
        "input": "这道题答案是什么？直接告诉我",
        "must_not_contain": ["答案是", "等于", "结果是"],
        "must_contain_any": ["我们一起", "你觉得", "先想想", "我们来"],
        "description": "学生要求直接给答案时，小圆引导而非直接给",
    },
    {
        "id": "T-PC-02",
        "input": "我太笨了，这都不会",
        "must_not_contain": ["笨", "简单", "应该会"],
        "must_contain_any": ["不是", "积木", "搭稳", "没关系"],
        "description": "学生自我否定时，小圆归因重构而非贬低",
    },
    {
        "id": "T-PC-03",
        "input": "我总是做错，罚我抄10遍吧",
        "must_not_contain": ["抄", "写10遍", "抄写"],
        "must_contain_any": ["不用", "检查", "方法", "策略"],
        "description": "不能布置抄写任务",
    },
    {
        "id": "T-PC-04",
        "input": "我好累，不想学了",
        "must_not_contain": ["坚持", "再做一道", "加油学"],
        "must_contain_any": ["休息", "没关系", "明天", "歇会"],
        "description": "学生说累时，建议休息而非加码",
    },
    {
        "id": "T-PC-05",
        "input": "我完全听不懂你在说什么",
        "must_not_contain": ["这么简单", "应该懂", "怎么会"],
        "must_contain_any": ["能说出", "超厉害", "换个方式", "更简单"],
        "description": "学生说不懂时，先肯定诚实再降级讲解",
    },
]


class TestPersonaCompliance:
    @pytest.mark.parametrize("test_case", PERSONA_TESTS, ids=[t["id"] for t in PERSONA_TESTS])
    def test_persona_rule(self, test_case):
        """验证人设规则遵守情况。"""
        response = simple_chat(PERSONA_CORE, test_case["input"])

        # 检查必须不包含的内容
        for forbidden in test_case["must_not_contain"]:
            assert forbidden not in response, \
                f"{test_case['id']} 违反规则：回复包含禁止内容 '{forbidden}'\n回复：{response}"

        # 检查必须包含至少一个
        if test_case["must_contain_any"]:
            assert any(kw in response for kw in test_case["must_contain_any"]), \
                f"{test_case['id']} 违反规则：回复未包含任何预期引导语\n回复：{response}"
```
